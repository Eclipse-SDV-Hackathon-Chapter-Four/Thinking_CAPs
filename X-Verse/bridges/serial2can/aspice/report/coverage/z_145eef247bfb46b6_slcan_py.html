1"""Lawicell SLCAN line codec for classic CAN (CAN 2.0A/B, DLC 0..8). 

2 

3Host -> device and device -> host frames share one ASCII format: 

4 

5 tIIIL<data> standard data frame TIIIIIIIIL<data> extended data frame 

6 rIIIL standard remote frame RIIIIIIIIL extended remote frame 

7 

8each terminated by CR. A device may append a 4-hex-digit timestamp (``Z1``), 

9which is accepted and ignored. Devices answer commands with CR (ok), BELL 

10(error), ``z``/``Z`` (frame accepted) or a response such as ``V1010``. 

11""" 

12import re 

13 

14import can 

15 

16CR, BELL = b"\r", b"\x07" 

17CAN_MAX_DLEN = 8 

18SFF_MASK, EFF_MASK = 0x7FF, 0x1FFFFFFF 

19TIMESTAMP_DIGITS = 4 

20MAX_LINE = 1 + 8 + 1 + 2 * CAN_MAX_DLEN + TIMESTAMP_DIGITS # longest frame line 

21_HEX = re.compile(rb"^[0-9A-Fa-f]*$") 

22# Frame line letter <-> (extended identifier, remote frame). 

23_KINDS = {b"t": (False, False), b"T": (True, False), b"r": (False, True), b"R": (True, True)} 

24_LETTERS = {flags: letter.decode() for letter, flags in _KINDS.items()} 

25 

26 

27class SlcanError(ValueError): 

28 """A frame line that does not follow the SLCAN syntax.""" 

29 

30 

31def _check_encodable(message: can.Message) -> None: 

32 if message.is_error_frame or message.is_fd: 

33 raise SlcanError("error and CAN FD frames cannot be carried by SLCAN") 

34 if not 0 <= message.dlc <= CAN_MAX_DLEN: 34 ↛ 35line 34 didn't jump to line 35 because the condition on line 34 was never true

35 raise SlcanError(f"DLC {message.dlc} out of range") 

36 if not 0 <= message.arbitration_id <= (EFF_MASK if message.is_extended_id else SFF_MASK): 

37 raise SlcanError(f"identifier {message.arbitration_id:#x} out of range") 

38 if not message.is_remote_frame and len(message.data) < message.dlc: 38 ↛ 39line 38 didn't jump to line 39 because the condition on line 38 was never true

39 raise SlcanError("data shorter than DLC") 

40

42def encode(message: can.Message) -> bytes: 

43 """SLCAN line (with CR) for a classic CAN message; raises SlcanError.""" 

44 _check_encodable(message) 

45 extended, remote, dlc = message.is_extended_id, message.is_remote_frame, message.dlc 

46 ident = f"{message.arbitration_id:08X}" if extended else f"{message.arbitration_id:03X}" 

47 payload = "" if remote else bytes(message.data[:dlc]).hex().upper() 

48 return f"{_LETTERS[(extended, remote)]}{ident}{dlc}{payload}".encode() + CR 

49 

50 

51def decode(line: bytes) -> can.Message: 

52 """CAN message for one frame line (without terminator); raises SlcanError.""" 

53 kind = _KINDS.get(line[:1]) 

54 if kind is None: 

55 raise SlcanError(f"not a frame line: {line!r}") 

56 extended, remote = kind 

57 id_digits = 8 if extended else 3 

58 header = 1 + id_digits + 1 

59 if len(line) < header or not _HEX.match(line[1:]): 

60 raise SlcanError(f"malformed frame line: {line!r}") 

61 ident = int(line[1:1 + id_digits], 16) 

62 dlc = int(line[1 + id_digits:header], 16) 

63 if dlc > CAN_MAX_DLEN or ident > (EFF_MASK if extended else SFF_MASK): 

64 raise SlcanError(f"identifier or DLC out of range: {line!r}") 

65 body = 0 if remote else dlc * 2 

66 if len(line) not in (header + body, header + body + TIMESTAMP_DIGITS): # optional timestamp 

67 raise SlcanError(f"length does not match DLC: {line!r}") 

68 data = b"" if remote else bytes.fromhex(line[header:header + body].decode()) 

69 return can.Message(arbitration_id=ident, is_extended_id=extended, is_remote_frame=remote, 

70 dlc=dlc, data=data)

73class LineSplitter: 

74 """Splits a device byte stream into SLCAN tokens. 

75 

76 Yields each CR-terminated line without its CR (``b""`` is a bare OK) and 

77 each BELL as ``BELL``. LF is ignored. Lines longer than ``MAX_LINE`` are 

78 discarded up to the next terminator and reported once as ``None``. 

79 """ 

80 

81 def __init__(self) -> None: 

82 self._buffer = bytearray() 

83 self._overlong = False 

84 

85 def feed(self, data: bytes) -> list[bytes | None]: 

86 """All tokens completed by ``data``, in order; partial lines are kept.""" 

87 out: list[bytes | None] = [] 

88 for value in data: 

89 byte = bytes((value,)) 

90 if byte == b"\n": 

91 continue 

92 if byte in (CR, BELL): 

93 if byte == CR or self._buffer or self._overlong: 

94 out.append(None if self._overlong else bytes(self._buffer)) 

95 self._buffer.clear() 

96 self._overlong = False 

97 if byte == BELL: 

98 out.append(BELL) 

99 elif len(self._buffer) < MAX_LINE: 

100 self._buffer += byte 

101 else: 

102 self._overlong = True 

103 return out 

104 

105 

106def tokens(data: bytes) -> list[bytes | None]: 

107 """Convenience wrapper: all tokens of a complete byte string.""" 

108 return LineSplitter().feed(data)
