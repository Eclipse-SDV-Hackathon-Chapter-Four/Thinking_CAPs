from native_operator import ROOT,PRIVATE,extract,write
from score_sw_fabric.storage import validate_run_root
from pathlib import Path
import sys,json,shutil,hashlib
validate_run_root(ROOT)
label=sys.argv[1];assert label=='nativefix'
worker=Path.cwd();assert worker==Path(json.loads((PRIVATE/label/'run.json').read_text())['native_context']['cwd'])
state=json.loads((PRIVATE/'live-meter-final/meter.json').read_text())
entry=next(e for e in state['records'] if e['task']=='qualification:lifecycle-704-'+label)
assert entry['status']=='observed' and not state['stopped']
result,usage=extract(entry['ticket']);write(ROOT/(label+'-model-output.json'),result)
assert result['kind']=='agent_result' and result['native_ids']==['feat_arc_sta__lifecycle__cfg_params_static']
assert result['path']=='config/mw_com_config.bzl'
path=ROOT/'lifecycle'/result['path'];text=path.read_text();before=hashlib.sha256(text.encode()).hexdigest()
assert 0<len(result['replacements'])<=3
for rep in result['replacements']:
    assert set(rep)=={'old','new'} and isinstance(rep['old'],str) and rep['old'] and isinstance(rep['new'],str)
    assert text.count(rep['old'])==1 and rep['old']!=rep['new']
    text=text.replace(rep['old'],rep['new'],1)
path.write_text(text)
for destination in [worker/result['path'],ROOT/'native-target'/result['path']]:
    destination.parent.mkdir(parents=True,exist_ok=True);shutil.copy2(path,destination)
validate_run_root(ROOT)
write(ROOT/(label+'-applied.json'),{'origin':'native Fabro command applying exact Flash structured replacements','before_sha256':before,'after_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),'usage':usage})
print(json.dumps({'applied':label,'path':result['path']}))
