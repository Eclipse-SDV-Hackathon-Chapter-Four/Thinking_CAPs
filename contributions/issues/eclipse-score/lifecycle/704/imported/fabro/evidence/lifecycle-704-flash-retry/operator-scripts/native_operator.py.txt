def write(path,value,private=False):
    path.parent.mkdir(mode=0o700,parents=True,exist_ok=True)

def extract(number):
    instruction=PRIVATE/'live-meter-proof/instruction.json';pin=hashlib.sha256(instruction.read_bytes()).hexdigest();raw=(ROOT/'provider-records'/pin/f'response-{number:02d}.raw').read_bytes();text='';models=set();finish=[]
    for line in raw.splitlines():
        if not line.startswith(b'data: ') or line==b'data: [DONE]':continue
        value=json.loads(line[6:]);models.add(value.get('model'))
        for c in value.get('choices',[]):
            assert not c.get('delta',{}).get('tool_calls'),'UNEXPECTED_TOOL_CALL'
            text+=c.get('delta',{}).get('content') or ''
            if c.get('finish_reason'):finish.append(c['finish_reason'])
    assert models=={'deepseek-flash'},models
    assert finish==['stop'],finish
    return json.loads(text),reported_usage(raw,stream=True)

def run_stage(item):
    current=item['run_id'];label=item['label']
    api('POST','/runs/'+current+'/start',{})
    start=time.monotonic()
    print(json.dumps({'started':label,'run_id':current}),flush=True)
    while time.monotonic()-start<240:
        validate_run_root(ROOT)
        value=api('GET','/runs/'+current)
        kind=value['lifecycle']['status']['kind']
        if kind in ('succeeded','failed','canceled','blocked'):break
        time.sleep(2)
    else:
        api('POST','/runs/'+current+'/cancel',{});raise RuntimeError('NATIVE_TIMEOUT')
    for name,suffix in [('inspect',''),('events','/events?limit=1000'),('stages','/stages')]:write(ROOT/(label+'-'+name+'.json'),api('GET','/runs/'+current+suffix))
    print(json.dumps({'finished':label,'status':kind}),flush=True)
    if label!='critique':assert kind=='succeeded',value['lifecycle']['status']
    return kind
